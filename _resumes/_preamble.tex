% assumed: `pdftotext -raw foo.pdf -` returns the two-column heading rows

% after both degrees -- which is why Workday's prefilled dates want a manual
% check. `pdffonts foo.pdf` shows two Type 3 bitmap fonts (the $ glyph and the

% \usepackage{lmodern} above removes them (tested: Type 3 2 -> 0, page count

% Custom commands
% The trailing \vspace used to sit INSIDE the item text, in horizontal mode and
% preceded by a space. When an item's last line ran close to the right margin,
% that space plus the spacing command wrapped onto a line of their own and left a
% visible blank line inside the list. Applying the skip after the item content,
% in vertical mode, gives the same tightening with no stray line.
\newcommand{\resumeItem}[1]{
  \item\small{#1}\vspace{-2pt}
}

\newcommand{\resumeItemBold}[2]{
  \item\small{\textbf{#1}{: #2}}\vspace{-2pt}
}

% A heading never ends a page, and an entry's bullets stay together. Each
% heading closes with \nopagebreak and each bullet list sets beginpenalty and
% midpenalty to 10000, so neither the glue after a heading, the start of its
% bullets, nor the space between two bullets is a legal break: a page breaks
% between entries. This only removes break points, so a resume that fits on one
% page lays out exactly as before.
% Just in case someone needs a heading that does not need to be in a list
\newcommand{\resumeHeading}[4]{
    \begin{tabular*}{0.99\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\nopagebreak\vspace{-5pt}
}

\newcommand{\resumeSubheading}[4]{
  \vspace{-1pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\nopagebreak\vspace{-5pt}
}

\newcommand{\resumeSubSubheading}[2]{
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \textit{\small\textbf{#1}} & \textit{\small #2} \\
    \end{tabular*}\nopagebreak\vspace{-5pt}
}

\newcommand{\resumeSubItem}[2]{\resumeItemBold{#1}{#2}\vspace{-4pt}}

% One-line italic company/role context under a subheading, above its bullets.
\newcommand{\resumeBlurb}[1]{

}

% A publication: one citation line -- authors, quoted title, italic venue --
% with an optional trailing link. Same \item as the other subheadings so it
% sits in the same list with the same marker; no two-column tabular, because a
% citation wraps as prose rather than pairing a left and a right cell.
\newcommand{\resumePublication}[4]{
  \vspace{-1pt}\item\small{#1. ``#2.'' \emph{#3}.#4}\vspace{-5pt}
}

\newcommand{\resumeCourseheading}[2]{

      \textbf{#1} & \textit{\small #2} \\
    \end{tabular*}\nopagebreak\vspace{-5pt}
}

% A labelled one-liner, deliberately NOT a \section: the CV's Research Interests
% line, Relevant Coursework under Education, and a variant's interests row. A
% ruled heading would give a single line the weight of a whole section. Flush
% left like the rest of the body, small-caps label; the hanging indent is
% measured from the label, so a wrapped second line aligns under the text
% rather than under the label. It only works outside a list (a list's
% \parshape overrides \hangindent), so call it between sections or after a
% list ends. The optional first argument is the space above it; the default
% keeps already-built variant .tex files, which pass two arguments, unchanged.
\newlength{\resumeInterestsIndent}
\newcommand{\resumeInterests}[3][9pt]{
  \vspace{#1}
  \settowidth{\resumeInterestsIndent}{\small\scshape #2\hspace{1.4em}}%
  \par\noindent
  \hangindent=\resumeInterestsIndent\hangafter=1%
  {\small\scshape #2}\hspace{1.4em}{\small #3}\par
  \vspace{3pt}
}

% The page-fill line the build reads from the log (_lib/pdflatex.js): the last
% page's text at its natural height (used), how much its spacing can shrink
% (shrink), how much a page holds (goal), and the height of one line of \small
% text (bullets) and of body text (a letter's paragraphs), all in points. A
% page fits while used minus shrink is at most goal, which is TeX's own test.
% The penalty is a legal break point, so TeX hands it the text so far and, if
% that runs past the page, ships every full page first, breaking where the
% \clearpage that ends the document would have. Without it the last break
% waits for that \clearpage, and the line would measure every page as one.
\AtEndDocument{%
  \par\penalty0
  \begingroup\small\xdef\resumeSmallLine{\the\baselineskip}\endgroup
  \begingroup\normalsize\xdef\resumeNormalLine{\the\baselineskip}\endgroup
  \typeout{RESUMEFILL used=\the\pagetotal\space shrink=\the\pageshrink\space
    goal=\the\pagegoal\space small=\resumeSmallLine\space normal=\resumeNormalLine}%
}
