% Custom commands
% The trailing \vspace used to sit INSIDE the item text, in horizontal mode and
% preceded by a space. When an item's last line ran close to the right margin,
% that space plus the spacing command wrapped onto a line of their own and left a
% visible blank line inside the list. Applying the skip after the item content,
% in vertical mode, gives the same tightening with no stray line.
\newcommand{\resumeItem}[1]{
  \item\small{#1}\vspace{-2pt}
}

\newcommand{\resumeItemBold}[2]{
  \item\small{\textbf{#1}{: #2}}\vspace{-2pt}
}

% Just in case someone needs a heading that does not need to be in a list
\newcommand{\resumeHeading}[4]{
    \begin{tabular*}{0.99\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-5pt}
}

\newcommand{\resumeSubheading}[4]{
  \vspace{-1pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-5pt}
}

\newcommand{\resumeSubSubheading}[2]{
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}

}

\newcommand{\resumeCourseheading}[2]{

    \end{tabular*}\vspace{-5pt}
}

% Interests: a one-liner near the top, deliberately NOT a \section -- it is a
% short personal aside, and a ruled heading would hand it the same weight as
% Professional Experience. Flush left like the rest of the body (a centered line
% here reads as a stray caption), small-caps label, picks separated by middots,
% with space above and below so it doesn't look like the summary's last line.
% The hanging indent is measured from the label, so a wrapped second line aligns
% under the picks rather than under the heading.
\newlength{\resumeInterestsIndent}
\newcommand{\resumeInterests}[2]{
  \vspace{9pt}
  \settowidth{\resumeInterestsIndent}{\small\scshape #1\hspace{1.4em}}%
  \par\noindent
  \hangindent=\resumeInterestsIndent\hangafter=1%
  {\small\scshape #1}\hspace{1.4em}{\small #2}\par
  \vspace{3pt}
}
