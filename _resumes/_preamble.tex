% Custom commands
% The trailing \vspace used to sit INSIDE the item text, in horizontal mode and
% preceded by a space. When an item's last line ran close to the right margin,
% that space plus the spacing command wrapped onto a line of their own and left a
% visible blank line inside the list. Applying the skip after the item content,
% in vertical mode, gives the same tightening with no stray line.
% An entry stays in one piece: a page breaks between entries, never inside one.
% TeX may break a page at a penalty below 10000, and at glue that directly
% follows a box, so every such place inside an entry is closed:
%   - a heading ends with \nopagebreak, ahead of its trailing \vspace;
%   - a bullet does too (below), since its \vspace sits right after its last
%     line, and a penalty placed after that glue would come too late;
%   - a bullet list sets beginpenalty, midpenalty and endpenalty to 10000, so
%     neither its start, the space between two bullets, nor its end is a break.
%     The end matters for a role with subprojects: the subproject's heading
%     follows the role's own bullets, and stays under its role.
%   - \interlinepenalty keeps the lines of one bullet together. With entries as
%     blocks, TeX would otherwise split a three-line bullet before it left a gap.
% The break between entries is the next entry's own \item, and a section's end
% is its list's. All of this only removes break points, so a resume that fits
% on one page lays out exactly as before.
\interlinepenalty=10000

\newcommand{\resumeItem}[1]{
  \item\small{#1}\nopagebreak\vspace{-2pt}
}

% Heading rows span the full line (\tabcolsep is 0 and the entry list has no
% margin), so the dates and places end where the section rule does.
\newcommand{\resumeSubheading}[4]{
  \vspace{-1pt}\item
    \begin{tabular*}{\linewidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\nopagebreak\vspace{-5pt}
}

\newcommand{\resumeSubSubheading}[2]{
    \begin{tabular*}{\linewidth}{l@{\extracolsep{\fill}}r}
      \textit{\small\textbf{#1}} & \textit{\small #2} \\
    \end{tabular*}\nopagebreak\vspace{-5pt}
}

% One-line italic company/role context under a subheading, above its bullets.
\newcommand{\resumeBlurb}[1]{
  \par\vspace{2pt}{\small\textit{#1}}\nopagebreak\vspace{-4pt}
}

% A publication: one citation line -- authors, quoted title, italic venue --
% with an optional trailing link. Same \item as the other subheadings so it
% sits in the same list; no two-column tabular, because a citation wraps as
% prose rather than pairing a left and a right cell.
\newcommand{\resumePublication}[4]{
  \vspace{-1pt}\item\small{#1. ``#2.'' \emph{#3}.#4}\nopagebreak\vspace{-5pt}
}

% A labelled one-liner, deliberately NOT a \section: the CV's Research Interests
% line, Relevant Coursework under Education, and a variant's interests row. A
% ruled heading would give a single line the weight of a whole section. Flush
% left like the rest of the body, small-caps label; the hanging indent is
% measured from the label, so a wrapped second line aligns under the text
% rather than under the label. It only works outside a list (a list's
% \parshape overrides \hangindent), so call it between sections or after a
% list ends. The optional first argument is the space above it; the default
% keeps already-built variant .tex files, which pass two arguments, unchanged.
% The \par sits inside the \small group, so a wrapped line gets \small's line
% spacing and the gap after the label is the one \settowidth measured.
\newlength{\resumeInterestsIndent}
\newcommand{\resumeInterests}[3][9pt]{
  \vspace{#1}
  \settowidth{\resumeInterestsIndent}{\small\scshape #2\hspace{1.4em}}%
  \par\noindent
  \hangindent=\resumeInterestsIndent\hangafter=1%
  {\small{\scshape #2}\hspace{1.4em}#3\par}
  \vspace{3pt}
}

% Entry headings start flush left under the section rule, with no marker of

\newcommand{\resumeBullet}{$\vcenter{\hbox{\tiny$\bullet$}}$}
\newcommand{\resumeItemListStart}{\begin{itemize}[label=\resumeBullet, beginpenalty=10000, midpenalty=10000, endpenalty=10000]}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{-5pt}}
% Skills: one group to a line in bullet-size text, its label run in ahead of
% its skills ("Languages: Python, ..."), with a point of air between groups. A
% table lined the skills up more neatly, but a text extractor that reads the
% page in blocks (pdftotext without -raw) took its label column as one block
% and lost which skills went with which label, and CMU's guide says to avoid
% tables for that reason. A group too long for its line hangs 1em in.
\newenvironment{resumeSkills}{%
  \begin{itemize}[leftmargin=1em, itemindent=-1em, label={}, itemsep=1pt, parsep=0pt]\small
}{%
  \end{itemize}
}

% The page-fill line the build reads from the log (_lib/pdflatex.js): the last
% page's text at its natural height (used), how much its spacing can shrink
% (shrink), how much a page holds (goal), and the height of one line of \small
% text (bullets) and of body text (a letter's paragraphs), all in points. A
% page fits while used minus shrink is at most goal, which is TeX's own test.
% The penalty is a legal break point, so TeX hands it the text so far and, if
% that runs past the page, ships every full page first, breaking where the
% \clearpage that ends the document would have. Without it the last break
% waits for that \clearpage, and the line would measure every page as one.
%
% Every page also writes a line of its own as it ships: how tall its text is at
% its natural height (used) and how much the page holds (goal). The fill line
% above only sees the last page, and an entry that doesn't fit moves to the
% next page whole, so an earlier page can end short with nothing to say so.
% \pagetotal can't tell: when the page ships it still holds the total from the
% moment the page overflowed, which is past the break TeX chose. So this
% measures the page as broken, box 255, before LaTeX builds the column from it.
\makeatletter
\let\resume@makecol\@makecol
\def\@makecol{%
  \begingroup\setbox\z@\vbox{\unvcopy\@cclv}%
  \typeout{RESUMEPAGE page=\the\c@page\space used=\the\ht\z@\space goal=\the\@colht}%
  \endgroup
  \resume@makecol}
\makeatother

  \begingroup\normalsize\xdef\resumeNormalLine{\the\baselineskip}\endgroup

    goal=\the\pagegoal\space small=\resumeSmallLine\space normal=\resumeNormalLine}%
}
